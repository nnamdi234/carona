async def create_ride(db: AsyncSession, user_input: RideCreate, current_user: User):
    # 1. the vehicle must exist
    result = await db.execute(
        select(Vehicle).where(Vehicle.id == user_input.vehicle_id)
    )
    vehicle = result.scalar_one_or_none()

    if vehicle is None:
        raise HTTPException(
            status_code=404,
            detail="Vehicle not found"
        )

    # 2. ...and belong to the person offering the ride
    if vehicle.owner_id != current_user.id:
        raise HTTPException(
            status_code=403,
            detail="You can only offer rides in your own vehicle"
        )

    # 3. you can't offer more seats than the car has
    if user_input.seats_offered > vehicle.capacity:
        raise HTTPException(
            status_code=400,
            detail="You can't offer more seats than your vehicle has"
        )

    # 4. the ride must be in the future
    if user_input.departure_time <= datetime.now(timezone.utc):
        raise HTTPException(
            status_code=400,
            detail="Departure time must be in the future"
        )

    new_ride = Ride(
        driver_id = current_user.id,
        vehicle_id = vehicle.id,
        origin = user_input.origin.strip(),
        destination = user_input.destination.strip(),
        departure_time = user_input.departure_time,
        seats_offered = user_input.seats_offered,
        fare = user_input.fare,
        status = "scheduled",
    )

    db.add(new_ride)
    await db.flush()
    await db.refresh(new_ride)

    return {"success": True, "message": "ride created successfully", "data": format_ride(new_ride, current_user, vehicle)}

async def search_rides(db: AsyncSession, origin: str | None, destination: str | None, day: date | None):
    # Ride + its driver + its vehicle, in one query
    query = (
        select(Ride, User, Vehicle)
        .join(User, Ride.driver_id == User.id)
        .join(Vehicle, Ride.vehicle_id == Vehicle.id)
        .where(Ride.status == "scheduled")
        .where(Ride.departure_time > datetime.now(timezone.utc))
    )

    # ilike = LIKE that ignores capital letters; % means "anything",
    # so "ikeja" finds "Ikeja City Mall"
    if origin:
        query = query.where(Ride.origin.ilike(f"%{origin.strip()}%"))

    if destination:
        query = query.where(Ride.destination.ilike(f"%{destination.strip()}%"))

    # a whole day in Lagos time: from 00:00 up to (not including) 00:00 the next day
    if day:
        start = datetime.combine(day, time.min, tzinfo=LAGOS)
        end = start + timedelta(days=1)
        query = query.where(Ride.departure_time >= start, Ride.departure_time < end)

    query = query.order_by(Ride.departure_time)

    result = await db.execute(query)
    rides = [format_ride(ride, driver, vehicle) for ride, driver, vehicle in result.all()]

    return {"success": True, "message": "rides retrieved successfully", "data": rides}

from sqlalchemy.orm import mapped_column
from sqlalchemy import String, Float, DateTime, Integer
from models.base import BaseModel
from sqlalchemy.dialects.postgresql import UUID
from sqlalchemy import ForeignKey


class Ride(BaseModel):

    __tablename__ = "rides"

    driver_id = mapped_column(
        UUID(as_uuid=True),
        ForeignKey("users.id", ondelete="CASCADE"),
        nullable=True,
    )

    vehicle_id = mapped_column(
        UUID(as_uuid=True),                 
        ForeignKey("vehicles.id", ondelete="CASCADE"),
        nullable=True
    )

    origin = mapped_column(String(255), nullable=False)

    destination = mapped_column(String(255), nullable=False)

    distance_km = mapped_column(Float, nullable=True)   

    departure_time =  mapped_column(
        DateTime(timezone=True),
        nullable=True,   
    )      

    arrival_time = mapped_column(       
        DateTime(timezone=True),
        nullable=True,   
    )

    status = mapped_column(String(100), nullable=False, default="scheduled")

    fare = mapped_column(Float, nullable=False)
    
    seats_offered = mapped_column(Integer, nullable=False)
